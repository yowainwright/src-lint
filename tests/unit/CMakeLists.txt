foreach(module imports config boundary graph)
  set(target "${module}_unit_test")
  add_executable("${target}" "${module}_test.c")
  target_link_libraries("${target}" PRIVATE src_lint_core)
  target_include_directories("${target}" PRIVATE "${PROJECT_SOURCE_DIR}/src")
  target_compile_definitions("${target}" PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options("${target}" PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(SRC_LINT_SANITIZERS)
    target_compile_options("${target}" PRIVATE ${SANITIZER_FLAGS})
  endif()
  add_test(NAME "unit_${module}" COMMAND "${target}")
  set_tests_properties("unit_${module}" PROPERTIES LABELS unit)
endforeach()
